前提：

- AssumeRole先：`OnPremSqlBridgeRole`
- リージョン：ap-northeast-1
- プロキシあり
- SQS（request / response）
- S3（結果保存）
- Oracle 18c（ODBC 64bit DSN使用）

利用サービス：

- Amazon Web Services
- AWS Security Token Service
- Amazon SQS
- Amazon S3

# 🚀 完全実装コード

```python
import os
import time
import json
import signal
import sys
import logging
import pyodbc
import boto3
from botocore.config import Config
from botocore.exceptions import ClientError
from datetime import datetime, timedelta

# =============================
# ログ設定
# =============================
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.FileHandler("onprem_sql_bridge.log"),
        logging.StreamHandler()
    ]
)

# =============================
# 設定
# =============================
AWS_REGION = os.getenv("AWS_REGION")
ROLE_ARN = os.getenv("ROLE_ARN")
REQUEST_QUEUE_URL = os.getenv("REQUEST_QUEUE_URL")
RESPONSE_QUEUE_URL = os.getenv("RESPONSE_QUEUE_URL")
RESULT_BUCKET = os.getenv("RESULT_BUCKET")

# アプリケーションID検証用
ALLOWED_APP_IDS = os.getenv("ALLOWED_APP_IDS", "").split(",")

# ORACLE_DSN はデフォルト接続先。tns 名は各メッセージから取得するため接続時に指定する。
ORACLE_DSN = os.getenv("ORACLE_DSN")
ORACLE_USER = os.getenv("ORACLE_USER")
ORACLE_PASSWORD = os.getenv("ORACLE_PASSWORD")
PROXY_URL = os.getenv("PROXY_URL")

SESSION_DURATION = 3600  # 秒
running = True


# =============================
# Graceful shutdown
# =============================
def shutdown_handler(signum, frame):
    global running
    logging.info("Shutdown signal received.")
    running = False

signal.signal(signal.SIGTERM, shutdown_handler)
signal.signal(signal.SIGINT, shutdown_handler)


# =============================
# STS AssumeRole
# =============================
def assume_role():
    logging.info("Assuming role...")
    base_session = boto3.Session()

    sts = base_session.client(
        "sts",
        region_name=AWS_REGION,
        config=Config(proxies={"https": PROXY_URL})
    )

    response = sts.assume_role(
        RoleArn=ROLE_ARN,
        RoleSessionName="OnPremSession",
        DurationSeconds=SESSION_DURATION
    )

    creds = response["Credentials"]

    expiration = creds["Expiration"]

    session = boto3.Session(
        aws_access_key_id=creds["AccessKeyId"],
        aws_secret_access_key=creds["SecretAccessKey"],
        aws_session_token=creds["SessionToken"],
        region_name=AWS_REGION
    )

    logging.info("Role assumed successfully.")

    return session, expiration


# =============================
# Oracle接続
# =============================
def connect_oracle(tns=None):
    """指定された tns 名でコネクションを作成（tns が None なら環境変数 DSN を使う）。"""
    dsn = tns or ORACLE_DSN
    logging.info(f"Connecting to Oracle DSN={dsn}...")
    conn = pyodbc.connect(
        f"DSN={dsn};UID={ORACLE_USER};PWD={ORACLE_PASSWORD}"
    )
    conn.autocommit = False
    return conn


# =============================
# SQL実行
# =============================
def validate_sql(sql: str) -> None:
    """簡易パーサーで SQL に禁止事項がないかチェック。"""
    s = sql.strip().lower()
    if s.startswith("select"):
        # MK_DATE 範囲指定が含まれているか確認。単純な正規表現で十分。
        if "mk_date" not in s or "where" not in s:
            raise ValueError("SELECT 文には WHERE 句と MK_DATE 範囲指定が必要です")
    elif s.startswith(("insert", "update")):
        # 単一レコード保証は実行後にチェックする
        pass
    else:
        raise ValueError("Unsupported SQL type")


def execute_sql(conn, sql):
    cursor = conn.cursor()

    if sql.strip().lower().startswith("select"):
        cursor.execute(sql)
        rows = cursor.fetchall()
        columns = [col[0] for col in cursor.description]
        result = [dict(zip(columns, row)) for row in rows]
        return "select", result

    elif sql.strip().lower().startswith(("insert", "update")):
        cursor.execute(sql)
        affected = cursor.rowcount
        conn.commit()
        if affected != 1:
            raise ValueError("INSERT/UPDATE は1行のみ許可されています")
        return "dml", affected

    else:
        raise Exception("Unsupported SQL type")


# =============================
# メイン処理
# =============================
def main():
    global running

    session, expiration = assume_role()
    sqs = session.client("sqs", config=Config(proxies={"https": PROXY_URL}))
    s3 = session.client("s3", config=Config(proxies={"https": PROXY_URL}))

    conn = connect_oracle()

    while running:

        # セッション期限チェック
        if datetime.utcnow() > expiration - timedelta(minutes=5):
            logging.info("Refreshing STS session...")
            session, expiration = assume_role()
            sqs = session.client("sqs", config=Config(proxies={"https": PROXY_URL}))
            s3 = session.client("s3", config=Config(proxies={"https": PROXY_URL}))

        try:
            response = sqs.receive_message(
                QueueUrl=REQUEST_QUEUE_URL,
                MaxNumberOfMessages=1,
                WaitTimeSeconds=10
            )

            messages = response.get("Messages", [])
            if not messages:
                continue

            for msg in messages:
                body = json.loads(msg["Body"])
                request_id = body.get("id")
                sql = body.get("sql")
                app_id = body.get("app_id")
                tns = body.get("tns")  # 接続先 DB 名

                # 必須フィールドチェック
                if not request_id or not sql or not app_id:
                    raise ValueError("message missing required fields")
                if app_id not in ALLOWED_APP_IDS:
                    raise ValueError(f"unauthorized app_id: {app_id}")

                # SQL 制約を検証
                validate_sql(sql)

                # DB 切り替えのため、毎回接続を張り直す（負荷低）
                if tns:
                    conn = connect_oracle(tns=tns)

                try:
                    sql_type, result = execute_sql(conn, sql)

                    if sql_type == "select":
                        key = f"{request_id}.json"
                        s3.put_object(
                            Bucket=RESULT_BUCKET,
                            Key=key,
                            Body=json.dumps(result),
                            ContentType="application/json"
                        )
                        payload = {"id": request_id, "s3_key": key}

                    else:
                        payload = {"id": request_id, "affected_rows": result}

                except Exception as e:
                    logging.error(f"SQL error: {e}")
                    conn.rollback()
                    payload = {"id": request_id, "error": str(e)}

                sqs.send_message(
                    QueueUrl=RESPONSE_QUEUE_URL,
                    MessageBody=json.dumps(payload)
                )

                sqs.delete_message(
                    QueueUrl=REQUEST_QUEUE_URL,
                    ReceiptHandle=msg["ReceiptHandle"]
                )

        except ClientError as e:
            logging.error(f"AWS error: {e}")
            time.sleep(5)

        except Exception as e:
            logging.error(f"Unexpected error: {e}")
            time.sleep(5)

    logging.info("Shutting down gracefully.")


if __name__ == "__main__":
    main()
```
